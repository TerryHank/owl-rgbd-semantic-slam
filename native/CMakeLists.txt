cmake_minimum_required(VERSION 3.16)
project(owl_slam LANGUAGES C CXX)
find_package(RTABMap 0.23 REQUIRED COMPONENTS core)
find_package(Threads REQUIRED)
add_executable(owl_slam owl_slam.cpp)
target_compile_features(owl_slam PRIVATE cxx_std_17)
target_link_libraries(owl_slam PRIVATE rtabmap::core Threads::Threads)

option(BUILD_RKNN_SEGMENTATION "Build the Rockchip NPU segmentation executable" OFF)
if(BUILD_RKNN_SEGMENTATION)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn)
  find_path(RKNN_INCLUDE_DIR rknn_api.h REQUIRED)
  find_library(RKNN_RUNTIME rknnrt REQUIRED)
  add_executable(yoloe_rknn yolo26n-seg-rknn.cpp)
  target_compile_features(yoloe_rknn PRIVATE cxx_std_17)
  target_include_directories(yoloe_rknn PRIVATE "${RKNN_INCLUDE_DIR}")
  target_link_libraries(yoloe_rknn PRIVATE "${RKNN_RUNTIME}" ${OpenCV_LIBS} Threads::Threads)
endif()
